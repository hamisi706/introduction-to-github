1.2 Top 75% HIV Burden Countries Globally

# Select latest year
hiv_latest <- hiv_data %>%
  filter(`Series.Year` == max(`Series.Year`, na.rm = TRUE)) %>%
  arrange(desc(`Observation.Value`))

hiv_latest$cumsum_prop <- cumsum(hiv_latest$`Observation.Value`) / sum(hiv_latest$`Observation.Value`, na.rm = TRUE)

top_75_countries <- hiv_latest %>%
  filter(cumsum_prop <= 0.75)

# Visualization
hiv_trend_top <- hiv_data %>%
  filter(`Geographic.area` %in% top_75_countries$`Geographic.area`)

hiv_trend_top %>%
  ggplot(aes(x = `Series.Year`, y = `Observation.Value`, color = `Geographic.area`)) +
  geom_line() +
  theme_minimal() +
  labs(title = "Trend of HIV Cases in Top 75% Countries Globally",
       y = "People Living with HIV",
       x = "Year")

1.3 Top 75% HIV Burden within WHO Regions

# By WHO region (Regional group)
top_75_by_region <- hiv_data %>%
  group_by(`Regional.group`, `Geographic.area`) %>%
  summarise(total_cases = sum(`Observation.Value`, na.rm = TRUE)) %>%
  group_by(`Regional.group`) %>%
  arrange(desc(total_cases)) %>%
  mutate(cumsum_prop = cumsum(total_cases) / sum(total_cases, na.rm = TRUE)) %>%
  filter(cumsum_prop <= 0.75)
## `summarise()` has grouped output by 'Regional.group'. You can override using
## the `.groups` argument.
hiv_trend_region <- hiv_data %>%
  filter(`Geographic.area` %in% top_75_by_region$`Geographic.area`)

hiv_trend_region %>%
  ggplot(aes(x = `Series.Year`, y = `Observation.Value`, color = `Geographic.area`)) +
  geom_line() +
  facet_wrap(~`Regional.group`) +
  theme_minimal() +
  labs(title = "Trend of HIV Cases in Top 75% Countries by WHO Region",
       y = "People Living with HIV",
       x = "Year")

## 1.5 Findings

> **Findings Paragraph:**
>
> The analysis shows a strong association between higher levels of multidimensional poverty and increased HIV prevalence across countries. Access to electricity, clean drinking water, and sanitation were major factors impacting HIV burden. Random effects confirmed significant variability at country and year levels, suggesting the importance of tailored interventions.
